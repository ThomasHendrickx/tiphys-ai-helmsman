/* Derivation for M3-P9 fix round 1, CR-001.
 *
 * MECHANISM: a value is read with a DEFAULT and then COMPARED or TESTED,
 * without first establishing that the value was PRESENT. The default makes an
 * absent field indistinguishable from a present, differing one.
 *
 * This enumerates the mechanism over the SHIPPED SURFACE. The shipped surface
 * is `dist/`, per package.json's `files`, and `dist/` is compiled from `src/`
 * and `bin/` alone, so those two trees are the sources of everything shipped.
 *
 * Stage 1: every nullish-coalescing site.
 * Stage 2: of those, the ones whose defaulted value is READ OUT OF AN EXTERNAL
 *          DECODED RECORD (an index expression, i.e. a field of a document the
 *          program did not construct).
 * Stage 3: of stage 2, the ones whose defaulted value then FLOWS INTO A
 *          COMPARISON: ===, !==, <, >, a Map/Set key, .includes, .has, .indexOf,
 *          .sort, or a group/join used as an identity.
 */
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";

const ROOTS = ["src", "bin"];
const WINDOW = 10; /* lines after the site in which a use is looked for */

function walk(dir, out = []) {
  for (const name of readdirSync(dir).sort()) {
    const path = join(dir, name);
    if (statSync(path).isDirectory()) { walk(path, out); continue; }
    if (path.endsWith(".ts")) out.push(path);
  }
  return out;
}

const files = ROOTS.flatMap((r) => walk(r));

/* An EXTERNAL READ is an index expression or an optional-index expression:
   `x[y]`, `x?.[y]`, `x.record[y]`. A property access by fixed dotted name on a
   value the program built itself (`row.mode`, `result.stdout`) is NOT one, and
   is reported separately so the exclusion is visible rather than assumed. */
const EXTERNAL_READ = /(\?\.)?\[[^\]]+\]\s*\?\?/;
const COMPARE = /===|!==|\bMap\b|\bSet\b|\.set\(|\.get\(|\.has\(|\.includes\(|\.indexOf\(|\.sort\(|[^<>=!]<[^<=]|[^<>=!]>[^>=]|\.join\(/;

const stage1 = [];
for (const file of files) {
  const lines = readFileSync(file, "utf8").split("\n");
  lines.forEach((line, i) => {
    if (!line.includes("??")) return;
    stage1.push({ file, line: i + 1, text: line.trimEnd(), lines, index: i });
  });
}

const stage2 = stage1.filter((s) => EXTERNAL_READ.test(s.text));

/* A site is stage 3 if the DEFAULTED EXPRESSION is used comparatively on the
   same line, or (when the site binds a name) that name appears in a comparative
   position within WINDOW lines below. */
function bindingName(text) {
  const m = /(?:const|let|var)\s+([A-Za-z_$][\w$]*)\s*(?::[^=]+)?=/.exec(text);
  return m ? m[1] : undefined;
}
const stage3 = [];
for (const s of stage2) {
  let why;
  if (COMPARE.test(s.text)) why = "same line";
  else {
    const name = bindingName(s.text);
    if (name) {
      const re = new RegExp("\\b" + name.replace(/\$/g, "\\$") + "\\b");
      for (let k = s.index + 1; k <= Math.min(s.index + WINDOW, s.lines.length - 1); k += 1) {
        const l = s.lines[k];
        if (re.test(l) && COMPARE.test(l)) { why = `binding ${name} used at line ${k + 1}`; break; }
      }
    }
  }
  if (why) stage3.push({ ...s, why });
}

const arg = process.argv[2] ?? "all";
function dump(label, rows, withWhy = false) {
  console.log(`\n===== ${label}: ${rows.length} site(s) =====`);
  for (const r of rows) {
    console.log(`${r.file}:${r.line}: ${r.text.trim()}` + (withWhy ? `\n    -> ${r.why}` : ""));
  }
}
if (arg === "1" || arg === "all") dump("STAGE 1, every nullish-coalescing site in src/ and bin/", stage1);
if (arg === "2" || arg === "all") dump("STAGE 2, defaulted value read out of an EXTERNAL indexed record", stage2);
if (arg === "3" || arg === "all") dump("STAGE 3, stage 2 AND the defaulted value flows into a comparison", stage3, true);
console.log(`\ncounts: stage1=${stage1.length} stage2=${stage2.length} stage3=${stage3.length}`);

### Appendix A: the enumerator, verbatim

```
/*
 * M3-P9 fix round 2, THE DERIVATION.
 *
 * MECHANISM UNDER SEARCH: two strings are compared for EQUALITY or DISTINCTNESS
 * without a declared CANONICAL FORM, so two representations of one value read
 * as two different values (or two different values read as one).
 *
 * This is NOT round 1's derivation. Round 1 searched for "a defaulted read
 * (`?? ""`) that flows into a comparison". That is a search for a MISSING
 * VALUE. This searches for a PRESENT value whose REPRESENTATION was never
 * canonicalised, which is why round 1's script cannot find these sites: none
 * of them involves `??` at all.
 *
 * Usage: node derive-canonical.mjs <repo-root> [1|2|3|all]
 */
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join, relative } from "node:path";

const repo = process.argv[2];
const arg = process.argv[3] ?? "all";

/* The SHIPPED SURFACE in source terms. `dist/` is what package.json ships and
   it is BUILT FROM `src/`, so `src/` and `bin/` are the shipped surface's
   sources. `scripts/` is included because gate-registry.yaml names those
   commands and AGENTS.md instructs a consumer to run them, even though the
   arbitration records that `scripts/` is not in package.json `files`. */
const ROOTS = ["src", "bin", "scripts"];
const EXT = /\.(ts|mjs|js)$/;

function walk(dir, out) {
  for (const name of readdirSync(dir).sort()) {
    const path = join(dir, name);
    const st = statSync(path);
    if (st.isDirectory()) walk(path, out);
    else if (EXT.test(name)) out.push(path);
  }
  return out;
}

const files = [];
for (const root of ROOTS) {
  try { walk(join(repo, root), files); } catch { /* absent root */ }
}

/* ---------------- STAGE 1: every string-comparison site ---------------- */
/* A comparison is an equality/inequality operator, a string-keyed container
   lookup, or a string predicate whose answer is "are these the same text". */
const COMPARISON =
  /(!==|===|!=[^=]|[^=!<>]=[^=]|\.includes\(|\.has\(|\.indexOf\(|\.lastIndexOf\(|\.startsWith\(|\.endsWith\(|\.localeCompare\(|\.get\(|\.set\(|\bnew Set\(|\bnew Map\()/;
/* `=` alone is assignment; the character-class above already excludes `==`,
   `<=`, `>=` and `!=`, but a bare assignment still matches, so equality is
   required to be one of the real operators. */
const REAL_COMPARISON =
  /(!==|===|!=[^=]|\.includes\(|\.has\(|\.indexOf\(|\.lastIndexOf\(|\.startsWith\(|\.endsWith\(|\.localeCompare\(|\.get\(|\.set\(|\bnew Set\(|\bnew Map\()/;

/* ---------------- STAGE 2: at least one operand is EXTERNAL ------------ */
/* EXTERNAL means the value entered the program from outside it: a decoded
   document, a file, a directory listing, argv, env, or a child process. A
   value the program itself built from its own literals is not external. */
const EXTERNAL_SEED =
  /(decodeDocument|readOperatorPath|readFileSync|readdirSync|readContextDocument|JSON\.parse|parseDocument|\bparse\(|process\.argv|process\.env|spawnSync|execFileSync|asRecord|frontmatter|loadRegistry|loadManifest|readVerdict|\.stdout\b|\.stderr\b)/;
/* A record subscript is the shape every one of CR-001's four sites had: a key
   read out of a decoded document by name. */
const RECORD_SUBSCRIPT = /\w(\?)?\[\s*["'`]/;

/* ---------------- STAGE 3: no declared canonical form ------------------ */
/* A canonical form is any declared transformation applied to the operand
   before it is compared. If one of these appears on the line, or on the line
   that BOUND the operand, the site has a canonical form (which may still be
   the wrong one, but it is DECLARED, so it is not this mechanism). */
const CANONICALISER =
  /(\.normalize\(|\.toLowerCase\(|\.toUpperCase\(|\.toLocaleLowerCase\(|\.trim\(|canonical|Canonical|establishField|normalizeProse|\.replace\()/;

/* The mechanism is about TWO STRINGS. A comparison against `undefined`, `null`,
   a boolean, a number, or the result of `typeof` is a PRESENCE or TYPE test,
   not a text comparison, and no canonical form applies to it. Excluding these
   is what separates this derivation from round 1's, which was searching for
   presence tests and therefore wanted exactly the sites this drops. */
const NOT_TEXT =
  /(===|!==|==|!=)\s*(undefined|null|true|false|-?\d+(\.\d+)?|NaN)\b|(undefined|null|true|false|-?\d+(\.\d+)?)\s*(===|!==|==|!=)|\btypeof\b|\.length\s*(===|!==|<|>|<=|>=)|\bArray\.isArray\b|instanceof/;

function comparesText(line) {
  if (NOT_TEXT.test(line)) return false;
  /* At least one operand must be textual: a string literal, a template
     literal, or a call/identifier known to yield text. A comparison with no
     textual operand anywhere on the line is comparing objects or numbers. */
  return /["'`]/.test(line) || /\b(id|path|name|value|text|key|kind|status|dimension|authority|mode|phase|slug|label|title|ref|branch|sha|line)\b/i.test(line);
}

const stage1 = [];
for (const file of files) {
  const rel = relative(repo, file);
  const lines = readFileSync(file, "utf8").split("\n");
  /* Taint set: bindings in this file whose value came from outside. Seeded
     from EXTERNAL_SEED, then propagated to fixpoint so a value read from a
     document and passed through two locals is still recognised as external. */
  const tainted = new Set();
  for (let pass = 0; pass < 6; pass += 1) {
    for (const line of lines) {
      const bind = /^\s*(?:const|let|var)\s+([A-Za-z_$][\w$]*)\s*(?::[^=]*)?=(.*)$/.exec(line);
      if (bind === null) continue;
      const [, name, rhs] = bind;
      if (EXTERNAL_SEED.test(rhs) || RECORD_SUBSCRIPT.test(rhs) ||
          [...tainted].some((t) => new RegExp(`\\b${t}\\b`).test(rhs))) {
        tainted.add(name);
      }
    }
  }
  lines.forEach((text, index) => {
    const stripped = text.replace(/^\s*(\/\/|\*|\/\*).*$/, "");
    if (stripped.trim() === "") return;
    if (!REAL_COMPARISON.test(stripped)) return;
    if (!COMPARISON.test(stripped)) return;
    if (!comparesText(stripped)) return;
    stage1.push({ file: rel, line: index + 1, text: text.trim(), tainted, lines, index });
  });
}

const stage2 = stage1.filter((s) => {
  if (EXTERNAL_SEED.test(s.text) || RECORD_SUBSCRIPT.test(s.text)) return true;
  return [...s.tainted].some((t) => new RegExp(`\\b${t}\\b`).test(s.text));
});

const stage3 = [];
for (const s of stage2) {
  /* The operand's binding line counts as part of the site: canonicalising at
     the bind and comparing at the use is a DECLARED canonical form. Look back
     a 12-line window, the same window shape round 1's derivation used. */
  const window = s.lines.slice(Math.max(0, s.index - 12), s.index + 1).join("\n");
  if (CANONICALISER.test(window)) continue;
  stage3.push(s);
}

function dump(title, rows) {
  console.log(`\n=== ${title}: ${rows.length} site(s) ===`);
  for (const r of rows) console.log(`${r.file}:${r.line}: ${r.text}`);
}
if (arg === "1" || arg === "all") dump("STAGE 1, every string-comparison site in the shipped surface", stage1);
if (arg === "2" || arg === "all") dump("STAGE 2, stage 1 where an operand is EXTERNALLY SOURCED", stage2);
if (arg === "3" || arg === "all") dump("STAGE 3, stage 2 with NO declared canonical form", stage3);
console.log(`\ncounts: stage1=${stage1.length} stage2=${stage2.length} stage3=${stage3.length}`);
```

## THE CANONICAL FORM, DECLARED

It is declared in the code beside the comparison, at `canonicalScalar` in
`src/checks.ts`, not only here, because a canonical form recorded only in a work
history is a canonical form the next editor will not know about.

| question | answer |
|---|---|
| Unicode normalisation | **NFKC**, applied FIRST |
| case | **folded** to lower case, reversing round 1 |
| whitespace | runs collapse to one space, then trim |
| characters with no visual weight | **REFUSED**, not stripped |
| character set | **printable ASCII, U+0020 to U+007E**; anything else refused |

The order is load-bearing: NFKC runs BEFORE the ASCII gate, so a fullwidth
letter folds to its ASCII form and is then RECOGNISED as the same value, rather
than being refused as foreign. That is why `f<U+FF41>mily-a` produces "produced-by
value family-a occurs in 2 of the 2 verdicts" and not a refusal: the value is
readable and the honest answer is that the pair is correlated. Reverse the two
steps and the diagnosis gets worse while the exit code stays the same.

### Why the character set is restricted, which is the part that actually closes the class

Measured, node v26.6.0, one call to `String.prototype.normalize`:

| character | NFKC folds it to | closed by NFKC alone |
|---|---|---|
| U+FF41 FULLWIDTH LATIN SMALL LETTER A | U+0061 `a` | yes |
| U+00A0 NO-BREAK SPACE | U+0020 space | yes |
| `e` + U+0301 COMBINING ACUTE | U+00E9 | yes, composed |
| U+0430 CYRILLIC SMALL LETTER A | unchanged | **no** |
| U+2013 EN DASH | unchanged | **no** |
| U+200B ZERO WIDTH SPACE | unchanged | **no** |
| U+00AD SOFT HYPHEN | unchanged | **no** |
| U+FEFF BYTE ORDER MARK | unchanged | **no** |
| U+202E RIGHT-TO-LEFT OVERRIDE | unchanged | **no** |

**NFKC folds three of nine.** No Unicode normalisation form maps a cross-script
homoglyph onto its lookalike, because they are genuinely different characters
that merely render alike; that is what confusable detection is for, and it needs
a `confusables.txt` table this package does not carry and which would go stale
against every Unicode revision. A restricted character set needs no table and
cannot go stale. It is the only rule considered that closes the Cyrillic, the en
dash, the zero-width space and the soft hyphen with one line.

I also considered and rejected MIXED-SCRIPT detection (Unicode TR39's shape,
expressible with `\p{Script=...}` which Node does support). Measured:

```
$ node -e 'const re=/\p{Script=Cyrillic}/u;
> console.log("cyrillic a:", re.test(String.fromCodePoint(0x0430)));
> console.log("en dash   :", re.test(String.fromCodePoint(0x2013)));
> console.log("ZWSP      :", re.test(String.fromCodePoint(0x200B)));'
cyrillic a: true
en dash   : false
ZWSP      : false
```

It catches the Cyrillic homoglyph and would be more permissive of non-ASCII
identifiers, but it does NOT catch the en dash, which is Common script, nor the
zero-width space. A
rule that closes two of four members while looking principled is worse than a
blunt one that closes four, because it invites the belief that the class is
closed.

**What this refuses that a user might have meant.** A genuinely non-ASCII model
family name, or one containing a typographic dash. I judge that acceptable and
state the trade rather than hiding it: every real value in this repository's
fixtures, schemas and documentation is ASCII kebab-case; the sibling dimension
`framing` already carries the schema pattern `^[a-z0-9]+(-[a-z0-9]+)*$`, so two
of the three dimensions were already required to be lowercase ASCII tokens and
`produced-by` was the outlier; and a refusal here is LOUD, names the exact
codepoint and its position, and is recoverable in one edit, whereas the failure
it replaces is SILENT and authorises a merge.

### Refuse, or strip? The choice, and why

The brief asked this explicitly. **REFUSE.** Stripping the invisible characters
would also have been fail-closed and was a real option, so the reason matters:

A document carrying a zero-width space in a model-family id reads one way to a
human and another way to the program. Silently repairing it hands back a green
having never said that the document was built to deceive its reader. This file
already applies exactly that rule one screen up under SC-011, that "could not
look" must never print as "looked and fine"; "looked, and what I found was
crafted to read as something else" belongs on the same side of that line. A
refusal that names `U+200B at position 7` tells the person holding the file
about a character they cannot see. A silent strip tells them nothing and leaves
the next reader of the committed file equally blind.

### The direction argument, per site, including the one that goes the other way

This check REFUSES when two reviews are NOT distinct, so collapsing more strings
onto one form produces MORE refusals. Aggressive canonicalisation is the
fail-closed direction here; timid canonicalisation is what left the hole. Site
by site:

| site | what collapsing does | direction |
|---|---|---|
| per-dimension distinctness loop | more values compare equal, so more correlations found | **fail-closed** |
| `merge-authority` against the policy constant | more values match the delegated grant, so the check applies more often | **fail-closed** |
| `phase` join key | group grows, so more verdicts are compared | **fail-closed** |
| `kind` group membership | more documents load as verdicts | **fail-closed** |
| mode id lookup | the mode a charter names is found more often | neutral; the security decision is the `merge-authority` row above, which is fail-closed |
| `decorrelationTriple` membership test | more documents pass membership | **permissive**, and it is the only one |

The last row is declared in the code as well as here. It is accepted because it
wins an attacker nothing: membership only decides whether the check PROCEEDS,
and what it proceeds to compare is the COMMITTED group, which a non-committed
document is not a member of and does not change. Anyone wanting the comparison
to run can submit the committed file itself.
